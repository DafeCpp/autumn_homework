cmake_minimum_required(VERSION 3.10)
project(tarjan_visualization LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT TARGET Utils)
  add_subdirectory(../../lib "${CMAKE_CURRENT_BINARY_DIR}/lib")
endif()
if(NOT TARGET scc_algorithm)
  add_library(scc_algorithm ../scc/src/scc.cpp)
  target_include_directories(scc_algorithm PUBLIC ../scc/src)
  target_link_libraries(scc_algorithm PUBLIC Utils)
endif()
add_executable(tarjan_visualization ../scc/src/main.cpp)
target_link_libraries(tarjan_visualization PRIVATE scc_algorithm)
target_compile_definitions(tarjan_visualization PRIVATE SCC_TARJAN)

enable_testing()
find_package(GTest REQUIRED)
add_executable(tarjan_visualization_tests ../scc/src/test.cpp)
target_link_libraries(tarjan_visualization_tests PRIVATE scc_algorithm GTest::gtest_main)
include(GoogleTest)
gtest_discover_tests(tarjan_visualization_tests TEST_PREFIX "tarjan.")
